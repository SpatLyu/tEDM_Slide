\begin{frame}
  \begin{columns}[t!, onlytextwidth]
    \begin{column}{0.48\textwidth}
        \includegraphics[height=1\textheight]{figure/granger_causality.png}
    \end{column}
    
    \begin{column}{0.48\textwidth}
        \textbf{Granger Causality}
        \begin{align}
        Y(t) &= \sum_{j=1}^{p} \alpha_j Y(t-j) + \xi_1(t) \label{eq_gc1} \\
        Y(t) &= \sum_{j=1}^{p} \alpha_j Y(t-j) + \sum_{j=1}^{p} \beta_j X(t-j) + \xi_2(t) \label{eq_gc2}
        \end{align}

        \noindent
        A F-test is performed with \textcolor{red}{\textit{the null hypothesis of $Y(t)$ equals to model \eqref{eq_gc1}}} against \textit{the alternative hypothesis of $Y(t)$ equals to model \eqref{eq_gc2}}.

        \vspace{1em}

        \noindent
        We say that $X$ Granger causes $Y$ if we \textcolor{red}{reject the null hypothesis}.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}
  \begin{columns}[t!, onlytextwidth]
    \begin{column}{0.48\textwidth}
        \includegraphics[height=1\textheight]{figure/transfer_entropy.png}
    \end{column}
    
    \begin{column}{0.48\textwidth}
        \textbf{Transfer Entropy}
        \begin{align*}
        H(X) &= -\sum_{x \in X} p(x) \log_2 p(x) \\
        T_{Y \to X}(t) &= I(X_t : Y_{t-1} | X_{t-1}) \\
                    &= H(X_t | X_{t-1}) - H(X_t | X_{t-1} Y_{t-1}') \\
                    &= \sum p(x_t, x_{t-1}, y_{t-1}) * \log_2 \frac{p(x_t | x_{t-1} y_{t-1}')}{p(x_t | x_{t-1})}
        \end{align*}

        \begin{itemize}
            \item Transfer entropy can be considered a \textcolor{red}{\textbf{non-parametric equivalent}} of Granger Causality (it also works for \textit{nonlinear categorical variables}).
            \item The \textcolor{red}{\textbf{mutual information}} between both is \textcolor{red}{\textbf{symmetric}} (undirected), but the introduced \textcolor{red}{\textbf{time delay}} allows for establishing directionality.
        \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{EDM Methods}

\begin{frame}
\begin{columns}[t!, onlytextwidth]
    \begin{column}{0.625\textwidth}
        \begin{figure}
        \centering
        \includegraphics[height=0.95\textheight]{figure/edm.png}
        \label{fig_edm}
        \end{figure}
    \end{column}
    
    \begin{column}[t!]{0.375\textwidth}
        \begin{block}{Empirical Dynamic Modeling (EDM)}
        Time-delay Embedding
        \begin{equation*}\label{eq_ts_embedding}
        \begin{split}
        M_{x} &= \left[ x(t),\, x(t-\tau),\, \ldots,\, x\big(t-(E - 1)\tau\big) \right] \\
        M_{y} &= \left[ y(t),\, y(t-\tau),\, \ldots,\, y\big(t-(E - 1)\tau\big) \right]
        \end{split}
        \end{equation*}

		Mutual Cross Mapping
        \begin{equation*}\label{eq_cross_mapping}
        \hat{X}_t \mid M_y = \sum_{i=1}^k \left( \omega_{ti} \, X_{ti} \mid M_y \right)
        \end{equation*}
        \begin{equation*}\label{eq_ccm_rho}
        \rho = \frac{\mathrm{Cov}\left( X, \hat{X} \mid M_y \right)}{ \sqrt{\mathrm{Var}(X) \, \mathrm{Var}\left( \hat{X} \mid M_y \right) } },
        \end{equation*}
    \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}
\begin{columns}[t!, onlytextwidth]
    \begin{column}{0.575\textwidth}
        \begin{figure}
        \centering
        \includegraphics[height=0.95\textheight]{figure/pcm.png}
        \label{fig_pcm}
        \end{figure}
    \end{column}
    
    \begin{column}[t!]{0.425\textwidth}
        \begin{block}{Partial Cross Mapping (PCM)}
        Accounting for indirect causal influences mediated by third variables
        \begin{equation*}\label{eq_pcm_zhat}
        \hat{Z}_t \mid M_y = \sum_{i=1}^k \left( \omega_{ti} \, Z_{ti} \mid M_y \right)
        \end{equation*}
        \begin{equation*}\label{eq_pcm_x_zhat}
        \hat{X}^{(Z)}_t \mid M_{\hat{Z}} = \sum_{i=1}^k \left( \omega^{(Z)}_{ti} \, X_{ti} \mid M_{\hat{Z}} \right)
        \end{equation*}
        \begin{equation*}\label{eq_pcm_partial_corr}
        \rho_{X \to Y \mid Z} = \frac{ \rho_{X,\, \hat{X}} - \rho_{X,\, \hat{X}^{(Z)}} \rho_{\hat{X},\, \hat{X}^{(Z)}} }
        {\sqrt{\left( 1 - \rho_{X,\, \hat{X}^{(Z)}}^2 \right) \left( 1 - \rho_{\hat{X},\, \hat{X}^{(Z)}}^2 \right) }}
        \end{equation*}
    \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}
\begin{columns}[t!, onlytextwidth]
    \begin{column}{0.65\textwidth}
        \begin{figure}
        \centering
        \includegraphics[height=0.9\textheight]{figure/cmc.png}
        \label{fig_cmc}
        \end{figure}
    \end{column}
    
    \begin{column}[t!]{0.35\textwidth}
        \begin{block}{Cross Mapping Cardinality (CMC)}
        Employing neighborhood topological consistency in place of direct prediction
        \begin{equation*}\label{eq_cmc_ic_ik_delong}
        IC_{i,k} = \left| M_y^{nn}(i,k) \cap M_y^{nn}\big(M_x^{nn}(i,k),k\big) \right|
        \end{equation*}
        \begin{equation*}\label{eq_cmc_ic_k_delong}
        IC_k = \frac{1}{T} \sum_{i=1}^T IC_{i,k}
        \end{equation*}
        \begin{equation*}\label{eq_cmc_ic_curve_delong}
        IC_{\text{curve}} = \{ IC_k \mid k = 1, 2, \dots, T \}
        \end{equation*}
    \end{block}
    \end{column}
  \end{columns}
\end{frame}

        \begin{block}{Dynamical Causality under Latent Confounders (CIC)}
        \begin{itemize}
            \item Mathematically separates coupled dynamics into orthogonal common (confounder) and private subspaces.
                
            \item Accurately infers causal directions using \textbf{only two observed variables}

        \end{block}
